26int main() {
27 using namespace Bembel;
28 using namespace Eigen;
29
30 Bembel::IO::Stopwatch sw;
31
32 int polynomial_degree_max = 2;
33 int refinement_level_max = 3;
34 std::complex<double> wavenumber(2., 0.);
35
36 // Load geometry from file "sphere.dat", which must be placed in the same
37 // directory as the executable
38 Geometry geometry("sphere.dat");
39
40 // Define evaluation points for scattered field, sphere of radius 2, 100
41 // equispaced points.
42 MatrixXd gridpoints = Util::makeSphereGrid(2., 100);
43
44 // Define analytical solution using lambda function, in this case a dipole
45 // centered on 0, see Data.hpp
46 const std::function<VectorXcd(Vector3d)> fun = [wavenumber](Vector3d pt) {
47 return Data::Dipole(pt, wavenumber, Vector3d(0.2, 0.2, 0.2),
48 Vector3d(0., 0.1, 0.1));
49 };
50
51 std::cout << "\n" << std::string(60, '=') << "\n";
52 // Iterate over polynomial degree.
53 for (int polynomial_degree = 1; polynomial_degree < polynomial_degree_max + 1;
54 ++polynomial_degree) {
55 VectorXd error(refinement_level_max + 1);
56 // Iterate over refinement levels
57 for (int refinement_level = 0; refinement_level < refinement_level_max + 1;
58 ++refinement_level) {
59 sw.tic();
60 std::cout << "Degree " << polynomial_degree << " Level "
61 << refinement_level;
62 // Build ansatz space
63 AnsatzSpace<MaxwellSingleLayerOperator> ansatz_space(
64 geometry, refinement_level, polynomial_degree);
65
66 // Set up load vector
67 DiscreteLinearForm<RotatedTangentialTrace<std::complex<double>>,
68 MaxwellSingleLayerOperator>
69 disc_lf(ansatz_space);
70 disc_lf.get_linear_form().set_function(fun);
71 disc_lf.compute();
72
73 // Set up and compute discrete operator
74 DiscreteOperator<H2Matrix<std::complex<double>>,
75 MaxwellSingleLayerOperator>
76 disc_op(ansatz_space);
77 disc_op.get_linear_operator().set_wavenumber(wavenumber);
78 disc_op.compute();
79
80 // solve system
81 GMRES<H2Matrix<std::complex<double>>, IdentityPreconditioner> gmres;
82 gmres.compute(disc_op.get_discrete_operator());
83 gmres.set_restart(2000);
84 auto rho = gmres.solve(disc_lf.get_discrete_linear_form());
85
86 // evaluate potential
87 DiscretePotential<MaxwellSingleLayerPotential<MaxwellSingleLayerOperator>,
88 MaxwellSingleLayerOperator>
89 disc_pot(ansatz_space);
90 disc_pot.get_potential().set_wavenumber(wavenumber);
91 disc_pot.set_cauchy_data(rho);
92 auto pot = disc_pot.evaluate(gridpoints);
93
94 // compute reference, print time, and compute error
95 MatrixXcd pot_ref(gridpoints.rows(), 3);
96 for (int i = 0; i < gridpoints.rows(); ++i)
97 pot_ref.row(i) = fun(gridpoints.row(i));
98 error(refinement_level) = (pot - pot_ref).rowwise().norm().maxCoeff();
99 std::cout << " time " << std::setprecision(4) << sw.toc() << "s\t\t";
100 std::cout << error(refinement_level) << std::endl;
101 }
102
103 // estimate rate of convergence and check whether it is at least 90% of the
104 // expected value
105 assert(
106 checkRateOfConvergence(error.tail(3), 2 * polynomial_degree + 1, 0.9));
107
108 std::cout << std::endl;
109 }
110 std::cout << std::string(60, '=') << std::endl;
111
112 return 0;
113}
