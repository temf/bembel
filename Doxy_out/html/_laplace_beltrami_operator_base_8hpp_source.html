21template <typename Derived>
22class LaplaceBeltramiOperatorBase : public LocalOperatorBase<Derived> {
23 public:
24 LaplaceBeltramiOperatorBase() {}
25 template <class T>
26 void evaluateIntegrand_impl(const T &super_space, const SurfacePoint &p1,
27 const SurfacePoint &p2,
28 Eigen::MatrixXd *intval) const {
29 // get basic information
30 const unsigned int elements_per_direction =
31 (1 << super_space.get_refinement_level());
32 const double h = 1. / ((double)(elements_per_direction));
33
34 // compute surface measures from tangential derivatives
35 double x_kappa = p1.segment<3>(6).cross(p1.segment<3>(9)).norm();
36
37 // integrand without basis functions
38 double integrand = x_kappa * p1(2) / (h * h);
39
40 super_space.addScaledSurfaceGradientInteraction(intval, integrand, p1, p2);
41
42 return;
43 }
44};
