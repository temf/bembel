12#include <Bembel/IO>

14int main() {
15 using namespace Bembel;
16 // The VTKwriter sets up initial geomety information.
17 Eigen::VectorXd linspace = Eigen::VectorXd::LinSpaced(50, 0, 3);
18 VTKDomainExport writer(linspace, linspace, linspace);
19
20 Geometry geo("sphere.dat");
21
22 // Now we can add user defined data. There are different options.
23 // Exceptionally handy is a dataset that describes the distance to the
24 // geometry. This is useful for paraview visualizations, s.t. one can use the
25 // threashold feature to clip the domain. For this, we can just iterate over
26 // the elements of a ClusterTree. Usually the ClusterTree is hidden in an
27 // ansatz space and can be accest via the get_mesh() method. Try running the
28 // example_VTKSurfaceExport and visualise both files on top of each other.
29 ClusterTree mesh(geo, 5);
30 std::function<double(const Eigen::Vector3d&)> fun1 =
31 [&](const Eigen::Vector3d& point_in_space) {
32 double dist = 1000;
33 for (auto e = mesh.get_element_tree().cpbegin();
34 e != mesh.get_element_tree().cpend(); ++e) {
35 double new_dist = ((*e).midpoint_ - point_in_space).norm();
36 dist = dist > new_dist ? new_dist : dist;
37 }
38 return dist;
39 };
40
41 // One can also visualize vector fields.
42
43 std::function<Eigen::Vector3d(const Eigen::Vector3d&)> fun2 =
44 [](const Eigen::Vector3d& point_in_space) {
45 return Eigen::Vector3d(point_in_space(0),
46 .5 * point_in_space(2) * point_in_space(1),
47 point_in_space(0));
48 };
49
50 writer.addDataSet("Distance_to_sphere", fun1);
51 writer.addDataSet("Some_vector_field", fun2);
52
53 // Finally, we print to file.
54 writer.writeToFile("example.vts");
55 return 0;
56}
