34class logspace {

38 public:
39 logspace(t first, t base) : curvalue(first), base(base) {}
40
41 t operator()() {

48int main() {
49 using namespace Bembel;
50 using namespace Eigen;
51
52 const int refinement_level = 3;
53 const int polynomial_degree_max = 3;
54
55 const double f_min = 1e-9;
56 const double base = 3;
57 const int N = 40;
58 std::vector<double> frequencies;
59 std::generate_n(std::back_inserter(frequencies), N + 1,
60 logspace<double>(f_min, base));
61
62 std::cout << "f_min: " << f_min << std::endl;
63 std::cout << "f_max: " << frequencies[frequencies.size() - 1] << std::endl;
64 std::cout << "size: " << frequencies.size() << std::endl;
65
66 Geometry geometry("sphere.dat");
67
68 // Define evaluation points for scattered field, sphere of radius 2
69 MatrixXd gridpoints = Util::makeSphereGrid(2., 5);
70
71 std::cout << "\n" << std::string(60, '=') << "\n";
72 for (int polynomial_degree = 1; polynomial_degree < polynomial_degree_max + 1;
73 ++polynomial_degree) {
74 std::string filename = "Dipole_P" + std::to_string(polynomial_degree) +
75 "M" + std::to_string(refinement_level);
76
77 IO::Logger<10> logger(filename + ".txt", ",");
78 logger.file("f", "original_condition", "original_pw_error",
79 "stabilized_condition", "stabilized_pw_error");
80 std::cout << "Degree " << polynomial_degree << " Level " << refinement_level
81 << std::endl;
82 for (auto f = frequencies.begin(); f != frequencies.end(); ++f) {
83 std::cout << std::setprecision(1) << std::scientific << *f << "Hz\t";
84
85 // set up excitation
86 double omega = 2 * BEMBEL_PI * *f;
87 double wavenumber_re =
88 omega * std::sqrt(Constants::mu0 * Constants::eps0);
89 std::complex<double> wavenumber =
90 std::complex<double>(wavenumber_re, 0.0);
91
92 Vector3cd k(0, 0, wavenumber);
93 Vector3cd E0(1, 0, 0);
94 // Define analytical solution using lambda function, in this case a dipole
95 // centered on 0, see Data.hpp
96 const std::function<VectorXcd(Vector3d)> fun = [wavenumber](Vector3d pt) {
97 return Data::Dipole(pt, wavenumber, Vector3d(0.2, 0.2, 0.2),
98 Vector3d(0., 0.1, 0.1));
99 };
100
101 // set up vector valued ansatz space
102 AnsatzSpace<InductanceMatrix> ansatz_space_vector(
103 geometry, refinement_level, polynomial_degree);
104
105 // set up matrix
106 AugmentedEFIE<MatrixXcd, InductanceMatrix> augmented_efie(
107 ansatz_space_vector, geometry);
108 augmented_efie.set_wavenumber(wavenumber);
109 augmented_efie.set_omega(omega);
110 augmented_efie.compute();
111
112 // assemble the right hand side
113 AugmentedEFIEExcitation<RotatedTangentialTrace<std::complex<double>>,
114 InductanceMatrix>
115 excitation(ansatz_space_vector, augmented_efie.get_dofs_scalar());
116 excitation.get_linear_form().set_function(fun);
117 excitation.compute();
118
119 // prepare post processing
120 DiscreteLocalOperator<MassMatrixScalarDisc> mass_matrix(
121 augmented_efie.get_ansatz_space_mass());
122 mass_matrix.compute();
123 MatrixXcd M = MatrixXcd(mass_matrix.get_discrete_operator());
124 DiscreteOperator<MatrixXcd, HelmholtzSingleLayerOperator> cap_matrix(
125 augmented_efie.get_ansatz_space_scalar());
126 cap_matrix.get_linear_operator().set_wavenumber(wavenumber);
127 cap_matrix.compute();
128 MatrixXcd P = cap_matrix.get_discrete_operator() / Constants::eps0;
129 PartialPivLU<MatrixXcd> lu_P;
130 lu_P.compute(P);
131
132 // solve the system, compute error and condition number
133 double condition_original, condition_stabilized;
134 double error_original, error_stabilized;
135 for (auto simulation = 0; simulation < 2; ++simulation) {
136 bool stabilized = false;
137 if (simulation == 1) {
138 stabilized = true;
139 }
140
141 // apply stabilization
142 if (stabilized) {
143 augmented_efie.stabilize();
144 }
145
146 // solve system
147 PartialPivLU<MatrixXcd> lu;
148 lu.compute(augmented_efie.get_system_matrix());
149 VectorXcd u = lu.solve(excitation.get_excitation());
150 VectorXcd j = u.head(augmented_efie.get_dofs_vector());
151 VectorXcd phi = u.tail(augmented_efie.get_dofs_scalar());
152
153 // correct stabilization in right hand side
154 if (stabilized) {
155 j *= 1. / (omega * std::complex<double>(0., 1.) * Constants::mu0);
156 }
157
158 // post processing
159 VectorXcd q_post = lu_P.solve(-M * phi);
160
161 MixedEFIE<InductanceMatrix, HelmholtzSingleLayerOperator> efie(
162 ansatz_space_vector, augmented_efie.get_ansatz_space_scalar());
163
164 efie.set_omega(omega);
165 efie.set_wavenumber(wavenumber);
166 efie.set_current(j);
167 efie.set_charges(q_post);
168
169 auto pot = efie.evaluate(gridpoints);
170
171 // compute reference solution
172 MatrixXcd pot_ref(gridpoints.rows(), 3);
173 for (int i = 0; i < gridpoints.rows(); ++i)
174 pot_ref.row(i) = fun(gridpoints.row(i));
175 double error = (pot - pot_ref).rowwise().norm().maxCoeff();
176
177 // compute condition number
178 JacobiSVD<MatrixXcd> svd(augmented_efie.get_system_matrix());
179 double cond =
180 std::abs(svd.singularValues()(0) /
181 svd.singularValues()(svd.singularValues().size() - 1));
182
183 if (stabilized) {
184 std::cout << "\t\t(stabilized) Error:\t";
185 error_stabilized = error;
186 condition_stabilized = cond;
187 } else {
188 std::cout << "(original) Error:\t";
189 error_original = error;
190 condition_original = cond;
191 }
192
193 std::cout << error << "\tCondition number:\t" << cond << std::endl;
194 }
195 logger.file(*f, condition_original, error_original, condition_stabilized,
196 error_stabilized);
197 }
198 }
199 std::cout << std::string(60, '=') << std::endl;
200 return 0;
201}
