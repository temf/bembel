20inline Eigen::Matrix<double, Eigen::Dynamic, 3> makeTensorProductGrid(
21 Eigen::VectorXd X, Eigen::VectorXd Y, Eigen::VectorXd Z) {
22 const int maxX = std::max(X.rows(), X.cols());
23 const int maxY = std::max(Y.rows(), Y.cols());
24 const int maxZ = std::max(Z.rows(), Z.cols());
25 Eigen::Matrix<double, Eigen::Dynamic, 3> out(maxX * maxY * maxZ, 3);
26
27 for (int iz = 0; iz < maxZ; iz++)
28 for (int iy = 0; iy < maxY; iy++)
29 for (int ix = 0; ix < maxX; ix++) {
30 out.row(ix + iy * maxX + iz * maxY * maxX) =
31 Eigen::Vector3d(X(ix), Y(iy), Z(iz));
32 }
33 return out;
34}

36inline Eigen::Matrix<double, Eigen::Dynamic, 3> makeSphereGrid(
37 const double r, const int nSample,
38 const Eigen::Vector3d center = Eigen::Vector3d(0, 0, 0)) {
39 double goldenRatio = (1 + sqrt(5.0)) / 2.0;
40 Eigen::Matrix<double, Eigen::Dynamic, 3> out(nSample, 3);
41 for (int i = 0; i < nSample; ++i) {
42 double phi = acos(1 - 2.0 * (i + 0.5) / nSample);
43 double theta = 2 * BEMBEL_PI * (i + 0.5) / goldenRatio;
44 out.row(i) = (Eigen::Vector3d(r * cos(theta) * sin(phi),
45 r * sin(theta) * sin(phi), r * cos(phi)) +
46 center);
47 }
48 return out;
49}
