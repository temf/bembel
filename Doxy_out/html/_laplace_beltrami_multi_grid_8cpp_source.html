25// MG solver for the Laplace Beltrami with Neumann condition.
26int mmg(const Eigen::VectorXd &constant_one,
27 const Eigen::SparseMatrix<double> &S,
28 const Eigen::Matrix<double, Eigen::Dynamic, 1> &L,
29 const Eigen::Matrix<double, Eigen::Dynamic, 1> &f,
30 Eigen::Matrix<double, Eigen::Dynamic, 1> &x,
31 const std::vector<Eigen::SparseMatrix<double>> &Ps, int lvl,
32 double tol = 1e-6, int max_iters = 1000) {
33 // stiffness matrices
34 std::vector<Eigen::SparseMatrix<double>> Ss(lvl + 1);
35 // the stiffness matrices on each level
36 Ss[lvl] = S;
37 for (int i = lvl - 1; i >= 0; --i)
38 Ss[i] = Ps[i].transpose() * Ss[i + 1] * Ps[i];
39
40 // Multi Grid Part
41 int iter = 0;
42 // initialize the x with zeros
43 x.resize(f.size());
44 x.setZero();
45 Eigen::VectorXd res;
46 do {
47 // repetitively call multiplicativeMultiGrid()
48 Bembel::MG::multiplicativeMultiGrid(Ss, f, &x, Ps, lvl);
49 // minus regularization terms such that the integral of function represented
50 // by x is 0. L.dot(x) is integral of x and L.dot(constant_one) is integral
51 // of constant one function.
52 x = x - L.dot(x) / L.dot(constant_one) * constant_one;
53 res = f - S * x;
54 ++iter;
55 } while (res.norm() > tol && iter < max_iters);
56 return iter;
57}
