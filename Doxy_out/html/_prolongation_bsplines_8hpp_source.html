20template <typename Scalar>
21std::vector<Eigen::Triplet<Scalar>> toEigenTriplets(
22 Eigen::SparseMatrix<Scalar> &M) {
23 std::vector<Eigen::Triplet<Scalar>> v;
24 for (int i = 0; i < M.outerSize(); i++)
25 for (typename Eigen::SparseMatrix<Scalar>::InnerIterator it(M, i); it; ++it)
26 v.emplace_back(it.row(), it.col(), it.value());
27 return v;
28}

36std::tuple<Eigen::MatrixXd, unsigned int> nonZerosInfoRow(
37 const std::vector<double> &knots,
38 const std::vector<double> &uniform_refined_knots,
39 unsigned int polynomial_degree, unsigned int row_id,
40 unsigned int guess = 0) {
41 unsigned int d = guess;
42 const unsigned int knots_size = static_cast<unsigned int>(knots.size());
43 while (d + 1 < knots_size &&
44 uniform_refined_knots[row_id] >= knots[d]) ++d;
45 if (d > 0) --d;
46 Eigen::MatrixXd non_zero_alpha =
47 Eigen::MatrixXd::Ones(1, polynomial_degree + 1);
48 for (auto i = 0; i < polynomial_degree; ++i) {
49 Eigen::MatrixXd R = Eigen::MatrixXd::Zero(i + 1, i + 2);
50 for (auto j = 0; j < i + 1; ++j) {
51 R(j, j) = (knots[d + j + 1] - uniform_refined_knots[row_id + i + 1]) /
52 (knots[d + j + 1] - knots[d + j - i]);
53 R(j, j + 1) = 1.0 - R(j, j);
54 }
55 non_zero_alpha.block(0, 0, 1, i + 2) =
56 non_zero_alpha.block(0, 0, 1, i + 1) * R;
57 }
58
59 return std::make_tuple(non_zero_alpha, d);
60}

66Eigen::SparseMatrix<double> prolongateBsplines1D(
67 unsigned int polynomial_degree, unsigned int level,
68 unsigned int knotrepetition = 1) {
69 assert(level > 0 &&
70 "Level must be 1 or larger for build prolongation matrix from level-1 "
71 "to level");
72 std::vector<double> knots = Bembel::Spl::MakeUniformKnotVector(
73 polynomial_degree + 1, ((1 << (level - 1)) - 1), knotrepetition);
74 std::vector<double> uniform_refined_knots = Spl::MakeUniformKnotVector(
75 polynomial_degree + 1, ((1 << level) - 1), knotrepetition);
76 unsigned int num_basis = knots.size() - polynomial_degree - 1;
77 unsigned int num_fine_basis =
78 uniform_refined_knots.size() - polynomial_degree - 1;
79 typedef typename Eigen::Triplet<double> Triplet;
80 std::vector<Triplet> tripletList;
81 tripletList.reserve(num_fine_basis * (polynomial_degree + 1));
82 int d = 0;
83 for (int i = 0; i < num_fine_basis; ++i) {
84 Eigen::MatrixXd non_zero_alpha;
85 std::tie(non_zero_alpha, d) = nonZerosInfoRow(
86 knots, uniform_refined_knots, polynomial_degree, i, d);
87 for (int j = 0; j < polynomial_degree + 1; ++j) {
88 tripletList.push_back(
89 Triplet(i, d - polynomial_degree + j, non_zero_alpha(0, j)));
90 }
91 }
92 Eigen::SparseMatrix<double> P;
93 P.resize(num_fine_basis, num_basis);
94 P.setFromTriplets(tripletList.begin(), tripletList.end());
95 return P;
96}

102Eigen::SparseMatrix<double> prolongateBsplines2D(
103 unsigned int polynomial_degree, unsigned int level,
104 unsigned int knotrepetition = 1) {
105 assert(level > 0 &&
106 "Level must be 1 or larger for build prolongation matrix from level-1 "
107 "to level");
108 Eigen::SparseMatrix<double> P_ =
109 prolongateBsplines1D(polynomial_degree, level, knotrepetition);
110 Eigen::SparseMatrix<double> P = Eigen::kroneckerProduct(P_, P_);
111 return P;
112}

119Eigen::SparseMatrix<double> prolongateBsplinesMultiPatches2D(
120 unsigned int polynomial_degree, unsigned int level,
121 unsigned int num_patches, unsigned int knotrepetition = 1) {
122 assert(level > 0 &&
123 "Level must be 1 or larger for build prolongation matrix from level-1 "
124 "to level");
125 Eigen::SparseMatrix<double> P_ =
126 prolongateBsplines2D(polynomial_degree, level, knotrepetition);
127 std::vector<Eigen::Triplet<double>> triplets_ = toEigenTriplets(P_);
128 std::vector<Eigen::Triplet<double>> triplets;
129 Eigen::SparseMatrix<double> P(num_patches * P_.rows(),
130 num_patches * P_.cols());
131 for (auto i = 0; i < num_patches; ++i) {
132 for (auto entry : triplets_) {
133 triplets.push_back(Eigen::Triplet<double>(entry.row() + i * P_.rows(),
134 entry.col() + i * P_.cols(),
135 entry.value()));
136 }
137 }
138 P.setFromTriplets(triplets.begin(), triplets.end());
139 return P;
140}
